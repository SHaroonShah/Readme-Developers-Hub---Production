import React from 'react';

export const Card_1 = ({ children, href, icon, iconColor, title, tooltip }) => {
  const Tag = href ? 'a' : 'div';

  return (
    <Tag
      className={`
        rounded-md border border-gray-200 bg-white shadow-sm p-5 dark:bg-inherit dark:border-white/10
        ${href ? 'no-underline! text-inherit! hover:bg-gray-100 dark:hover:bg-gray-600' : ''}
      `}
      href={href}
      {...(tooltip ? { title: tooltip } : {})}
      {...(tooltip ? { 'aria-label': tooltip } : {})}
      {...(href ? { target: '_blank', rel: 'noopener noreferrer' } : {})}
    >
      {icon && (
        <i
          className={`text-xl fa-duotone fa-solid ${icon} mb-3`}
          style={iconColor ? { color: iconColor } : undefined}
        />
      )}
      {title && <p className="text-gray-800 dark:text-white font-semibold">{title}</p>}
      <div className="text-gray-600 dark:text-gray-300">{children}</div>
    </Tag>
  );
};

export const Cards_1 = ({ columns = 2, children }) => {
  return (
    <div
      className="grid gap-5 max-[414px]:grid-cols-1"
      style={{ gridTemplateColumns: `repeat(${Math.max(1, Number(columns) || 1)}, minmax(0, 1fr))` }}
    >
      {children}
    </div>
  );
};
