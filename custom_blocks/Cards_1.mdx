import React from 'react';

export const Card = ({ children, href, icon, iconColor, title }) => {
  const Tag = href ? 'a' : 'div';
  return (
    <Tag
      className={`
        rounded-md border border-gray-200 bg-white shadow-sm p-5 dark:bg-inherit dark:border-white/10
        ${href && 'no-underline! text-inherit! hover:bg-gray-100 dark:hover:bg-gray-600'}
      `}
      href={href}
      // Always open in a new tab if href exists
      {...(href ? { target: '_blank', rel: 'noopener noreferrer' } : {})}
    >
      {icon && <i className={`text-xl fa-duotone fa-solid ${icon} text-${iconColor} mb-3`}></i>}
      {title && <p className="text-gray-800 dark:text-white font-semibold">{title}</p>}
      <div className="text-600 dark:text-gray-300">{children}</div>
    </Tag>
  );
};

export const Cards = ({ columns = 2, children }) => {
  return (
    <div className={`grid gap-5 ${columns <= 1 ? 'grid-cols-1' : `grid-cols-${columns}`} max-[414px]:grid-cols-1`}>
      {children}
    </div>
  );
};

// Usage Example
<Cards columns={3}>
  <Card title="First Card" href="https://readme.com" icon="fa-home" iconColor="blue">
    *Lorem ipsum dolor sit amet, consectetur adipiscing elit*
  </Card>
  <Card title="Third Card" icon="fa-star" iconColor="blue-500">
    `Ut enim ad minim veniam, quis nostrud ullamco`
  </Card>
  <Card title="Fourth Card" icon="fa-question" iconColor="blue-500">
    **Excepteur sint occaecat cupidatat non proident**
  </Card>
</Cards>
