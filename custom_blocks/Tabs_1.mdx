import React, { useState } from 'react';

export const Tab = ({ children }) => {
  return <div>{children}</div>;
};

export const Tabs = ({ children, tabBorderGap = 13, activeBorderThickness = 3, activeBorderColor='#ca8a04', inactiveBorderColor='#e5e7eb' }) => {
  const [activeTab, setActiveTab] = useState(0);

  return (
    <div>
      <header>
        <nav className="flex gap-6 border-b border-gray-300 dark:border-gray-600 mb-5"
style={{ paddingBottom: tabBorderGap, borderBottom: `1px solid {#e5e7eb}`}}>
          {children?.map((tab, index) => {
            const isActive = activeTab === index;
            return (
              <button
                key={tab.props.title}
                className={`
                 bg border-1 font-semibold text-sm cursor-pointer mr-4 pb-
                 text-black hover:text-yellow-500 dark:hover:text-yellow-700 rounded
                 ${isActive && 'text-white-700 -mb-1 border-b-1 border-gray-600 dark:text-yellow-600 dark:border-yellow-500'}`}
                onClick={() => setActiveTab(index)}

									className={`  
									pb-1 text-sm font-semibold cursor-pointer
                  text-gray-700 dark:text-gray-300
                  hover:text-yellow-500 dark:hover:text-yellow-700
                  transition-colors 
 
                  ${isActive ? 'text-yellow-500 dark:text-yellow-500 border-b-2 border-yellow-500' : 'border-b-2 border-transparent'}
                `}

              >
                {tab.props.icon && (
                 <i className={`fa-duotone fa-solid ${tab.props.icon} text-${tab.props.iconColor} mr-2`}></i>
                )}
                {tab.props.title}

{isActive && (
                  <span
                    className="absolute left-0 right-0"
                    style={{
                      bottom: -tabBorderGap,
                      height: `${activeBorderThickness}px`,
                      backgroundColor: activeBorderColor,
                    }}
                    />
)}
              </button>
           )}
          )}
        </nav>
      </header>
      <section>{children && children[activeTab]}</section>
    </div>
  );
};

<Tabs>
  <Tab title="First Tab" icon="fa-clock" iconColor="blue-500">
    Welcome to the content that you can only see inside the first Tab.
  </Tab>
  <Tab title="Second Tab" icon="fa-star" iconColor="blue-500">
    This is content that you can only see inside the second Tab.
  </Tab>
  <Tab title="Third Tab" icon="fa-heart" iconColor="blue-500">
    This is content that you can only see inside the third Tab.
  </Tab>
</Tabs>
